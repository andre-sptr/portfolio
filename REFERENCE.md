            <script>
                function switchTab(tabId) {
                    const btnGithub = document.getElementById('tab-btn-github');
                    const btnCerts = document.getElementById('tab-btn-certs');
                    const radioGithub = document.getElementById('tab-radio-github');
                    const radioCerts = document.getElementById('tab-radio-certs');
                    const contentGithub = document.getElementById('content-github');
                    const contentCerts = document.getElementById('content-certs');

                    if (tabId === 'github') {
                        if (radioGithub) radioGithub.checked = true;
                        if (btnGithub) btnGithub.setAttribute('aria-selected', 'true');
                        if (btnCerts) btnCerts.setAttribute('aria-selected', 'false');
                        if (contentGithub) {
                            contentGithub.classList.remove('hidden');
                            contentGithub.classList.add('block');
                        }
                        if (contentCerts) {
                            contentCerts.classList.remove('block');
                            contentCerts.classList.add('hidden');
                        }
                    } else {
                        if (radioCerts) radioCerts.checked = true;
                        if (btnCerts) btnCerts.setAttribute('aria-selected', 'true');
                        if (btnGithub) btnGithub.setAttribute('aria-selected', 'false');
                        if (contentCerts) {
                            contentCerts.classList.remove('hidden');
                            contentCerts.classList.add('block');
                        }
                        if (contentGithub) {
                            contentGithub.classList.remove('block');
                            contentGithub.classList.add('hidden');
                        }
                    }
                }

                // Radio & label change handlers
                const radioGithub = document.getElementById('tab-radio-github');
                const radioCerts = document.getElementById('tab-radio-certs');
                const btnGithub = document.getElementById('tab-btn-github');
                const btnCerts = document.getElementById('tab-btn-certs');

                if (radioGithub) radioGithub.addEventListener('change', () => switchTab('github'));
                if (radioCerts) radioCerts.addEventListener('change', () => switchTab('certs'));
                if (btnGithub) btnGithub.addEventListener('click', () => switchTab('github'));
                if (btnCerts) btnCerts.addEventListener('click', () => switchTab('certs'));

                // Arrow key navigation between tabs
                document.querySelectorAll('#credentials-tab-track [role="tab"]').forEach(tab => {
                    tab.addEventListener('keydown', (e) => {
                        const tabs = Array.from(document.querySelectorAll('#credentials-tab-track [role="tab"]'));
                        const index = tabs.indexOf(e.target);
                        let newIndex;

                        if (e.key === 'ArrowRight' || e.key === 'ArrowDown') {
                            e.preventDefault();
                            newIndex = (index + 1) % tabs.length;
                        } else if (e.key === 'ArrowLeft' || e.key === 'ArrowUp') {
                            e.preventDefault();
                            newIndex = (index - 1 + tabs.length) % tabs.length;
                        }

                        if (newIndex !== undefined) {
                            tabs[newIndex].focus();
                            tabs[newIndex].click();
                        }
                    });
                });
            </script>

    <script>
        (function () {
            const detailsBtn = document.getElementById('details-btn');
            const detailsDropdown = document.getElementById('details-dropdown');
            const menuContainer = document.getElementById('details-menu-container');
            if (!detailsBtn || !detailsDropdown || !menuContainer) return;

            const glideContainer = document.getElementById('details-glide-container');
            const glidePill = document.getElementById('details-glide-pill');
            const menuItems = detailsDropdown.querySelectorAll('.details-menu-item');

            function moveGlidePill(item) {
                if (!glidePill || !item) return;
                const top = item.offsetTop;
                const height = item.offsetHeight;
                glidePill.style.transform = 'translateY(' + top + 'px)';
                glidePill.style.height = height + 'px';
                glidePill.classList.add('is-active');
            }

            function resetGlidePill() {
                if (!glidePill) return;
                glidePill.classList.remove('is-active');
            }

            if (menuItems.length) {
                menuItems.forEach(item => {
                    item.addEventListener('mouseenter', () => moveGlidePill(item));
                    item.addEventListener('focus', () => moveGlidePill(item));
                });
            }

            if (glideContainer) {
                glideContainer.addEventListener('mouseleave', resetGlidePill);
            }

            function toggleDetailsMenu(e) {
                if (e) e.stopPropagation();
                const isOpen = detailsDropdown.classList.contains('open');
                if (isOpen) {
                    closeDetailsMenu();
                } else {
                    openDetailsMenu();
                }
            }

            function openDetailsMenu() {
                detailsDropdown.classList.add('open');
                detailsBtn.setAttribute('aria-expanded', 'true');
            }

            function closeDetailsMenu() {
                detailsDropdown.classList.remove('open');
                detailsBtn.setAttribute('aria-expanded', 'false');
                resetGlidePill();
            }

            detailsBtn.addEventListener('click', toggleDetailsMenu);

            // Close when clicking an item
            menuItems.forEach(item => {
                item.addEventListener('click', () => {
                    closeDetailsMenu();
                });
            });

            // Close on click outside
            document.addEventListener('click', (e) => {
                if (!menuContainer.contains(e.target)) {
                    closeDetailsMenu();
                }
            });

            // Close on Escape key
            document.addEventListener('keydown', (e) => {
                if (e.key === 'Escape' && detailsDropdown.classList.contains('open')) {
                    closeDetailsMenu();
                    detailsBtn.focus();
                }
            });
        })();
    </script>

    <script>
        if ('scrollRestoration' in history) {

        const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
        const isMobileViewport = () => window.innerWidth < 1024;

        let lenis = null;

        function initLenis() {
            // Mobile isolation: Lenis is completely disabled on mobile (under 1024px)
            // Mobile devices use native 60fps/120fps hardware-accelerated touch momentum
            if (isMobileViewport() || prefersReducedMotion) {
                if (lenis) {
                    try {
                        lenis.stop();
                        lenis.destroy();
                    } catch (e) { }
                    lenis = null;
                    window.lenis = null;
                }
                return;
            }

            if (!lenis && typeof Lenis !== 'undefined') {
                lenis = new Lenis({
                    duration: 0.45,
                    easing: (t) => Math.min(1, 1.001 - Math.pow(2, -10 * t)),
                    direction: 'vertical',
                    gestureDirection: 'vertical',
                    smooth: true,
                    mouseMultiplier: 1.15,
                    smoothTouch: false,
                    touchMultiplier: 2,
                    infinite: false,
                });
                window.lenis = lenis;

                lenis.on('scroll', (e) => {
                    ScrollTrigger.update();
                    const scrollPos = typeof e.scroll === 'number' ? e.scroll : (window.scrollY || window.pageYOffset || 0);
                    updateNavbarOnScroll(scrollPos);
                    if (window._triggerScrollSpy) {
                        window._triggerScrollSpy(scrollPos);
                    }
                });
            }
        }

        initLenis();

        // High-precision native rAF loop for Lenis (direct vsync timing, zero stutter)
        let lenisRafId = 0;
        function lenisRafLoop(time) {
            if (!lenis || isMobileViewport() || document.hidden) {
                lenisRafId = 0;
                return;
            }
            lenis.raf(time);
            lenisRafId = requestAnimationFrame(lenisRafLoop);
        }
        function startLenisRaf() {
            if (lenis && !isMobileViewport() && !document.hidden && !lenisRafId) {
                lenisRafId = requestAnimationFrame(lenisRafLoop);
            }
        }
        startLenisRaf();
        document.addEventListener('visibilitychange', () => {
            if (document.hidden) {
                cancelAnimationFrame(lenisRafId);
                lenisRafId = 0;
            } else {
                startLenisRaf();
            }
        });

        // Navbar and Mobile Floating Nav scroll state
        let lastScroll = 0;
        const navbar = document.getElementById('navbar');
        const mobileLiquidNav = document.getElementById('mobile-liquid-nav');

        function updateNavbarOnScroll(currentScroll) {
            // Desktop navbar scroll behavior
            if (navbar && !isMobileViewport()) {
                if (currentScroll <= 0) {
                    navbar.style.transform = 'translateY(0)';
                } else if (currentScroll > lastScroll && currentScroll > 50) {
                    navbar.style.transform = 'translateY(-100%)';
                } else {
                    navbar.style.transform = 'translateY(0)';
                }
            }

            // Mobile floating button scroll behavior
            if (mobileLiquidNav && isMobileViewport()) {
                if (currentScroll <= 0) {
                    mobileLiquidNav.style.transform = 'translateY(0)';
                    mobileLiquidNav.style.opacity = '1';
                    mobileLiquidNav.style.pointerEvents = 'auto';
                } else if (currentScroll > lastScroll && currentScroll > 40) {
                    mobileLiquidNav.style.transform = 'translateY(-80px)';
                    mobileLiquidNav.style.opacity = '0';
                    mobileLiquidNav.style.pointerEvents = 'none';
                    if (typeof window._closeMobileLiquidNav === 'function') {
                        window._closeMobileLiquidNav();
                    }
                } else if (currentScroll < lastScroll) {
                    mobileLiquidNav.style.transform = 'translateY(0)';
                    mobileLiquidNav.style.opacity = '1';
                    mobileLiquidNav.style.pointerEvents = 'auto';
                }
            }

            lastScroll = currentScroll;
        }

        // Native scroll listener: drives mobile scroll events and fallback
        window.addEventListener('scroll', () => {
            if (isMobileViewport() || !lenis) {
                const scrollPos = window.scrollY || window.pageYOffset || 0;
                updateNavbarOnScroll(scrollPos);
                if (window._triggerScrollSpy) {
                    window._triggerScrollSpy(scrollPos);
                }
                
            }
        }, { passive: true });

        (function () {
            const nav = document.getElementById('mobile-liquid-nav');
            const btn = document.getElementById('mobile-liquid-btn');
            const btnText = document.getElementById('mobile-liquid-btn-text');
            const backdrop = document.getElementById('mobile-liquid-backdrop');
            if (!nav || !btn) return;

            function toggleMenu(e) {
                if (e) e.stopPropagation();
                const isOpen = nav.classList.toggle('is-open');
                btn.setAttribute('aria-expanded', isOpen ? 'true' : 'false');
                if (btnText) btnText.textContent = isOpen ? 'Close' : 'Details';
                if (backdrop) {
                    if (isOpen) backdrop.classList.add('is-active');
                    else backdrop.classList.remove('is-active');
                }
                if (isOpen && typeof window._closeMobileSocialDial === 'function') {
                    window._closeMobileSocialDial();
                }
            }

            function closeMenu() {
                if (nav.classList.contains('is-open')) {
                    nav.classList.remove('is-open');
                    btn.setAttribute('aria-expanded', 'false');
                    if (btnText) btnText.textContent = 'Details';
                    if (backdrop) backdrop.classList.remove('is-active');
                }
            }

            window._closeMobileLiquidNav = closeMenu;

            btn.addEventListener('click', toggleMenu);
            if (backdrop) backdrop.addEventListener('click', closeMenu);
            document.addEventListener('click', function (e) {
                if (!nav.contains(e.target)) closeMenu();
            });
            window.addEventListener('keydown', function (e) {
                if (e.key === 'Escape') closeMenu();
            });
        })();
    </script>
